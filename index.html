const body = { contents: [{ parts: [ { text: prompt }, { inlineData: { mimeType: mimeType || "audio/mp4", data: audioBase64 } }, ], }], generationConfig: { responseMimeType: "application/json", responseSchema: { type: "OBJECT", properties: { name: { type: "STRING" }, type: { type: "STRING" }, region: { type: "STRING" }, budget: { type: "STRING" }, pyeong: { type: "STRING" }, timing: { type: "STRING" }, memo: { type: "STRING" }, }, }, }, };
